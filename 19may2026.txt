USE VITBranches;

SHOW TABLES FROM VITBranches;

SELECT *
FROM Worker;

SELECT FIRST_NAME, DEPARTMENT, SALARY
FROM Worker
WHERE DEPARTMENT='Finance' AND SALARY>150000;

SELECT FIRST_NAME, DEPARTMENT, SALARY
FROM Worker
WHERE DEPARTMENT='IT' OR SALARY>250000;

SELECT WORKER_ID, FIRST_NAME, DEPARTMENT, SALARY
FROM Worker
WHERE WORKER_ID IN (101,104,106);

SELECT WORKER_ID, FIRST_NAME, DEPARTMENT, SALARY
FROM Worker
WHERE WORKER_ID NOT IN (102,103,105,107);

SELECT *
FROM Worker
WHERE SALARY BETWEEN 120000 AND 450000;

SELECT *
FROM Worker
WHERE SALARY NOT BETWEEN 70000 AND 100000;

SELECT *
FROM Worker
WHERE SALARY BETWEEN 100000 AND 400000
      AND DEPARTMENT NOT IN ('IT','Support');

SELECT *
FROM Worker
WHERE FIRST_NAME LIKE 'R%';

SELECT *
FROM Worker
WHERE FIRST_NAME LIKE '%a';

SELECT *
FROM Worker
WHERE FIRST_NAME LIKE '_e%';

SELECT *
FROM Worker
WHERE FIRST_NAME LIKE '%an%';

SELECT *
FROM Worker
WHERE DEPARTMENT='Management'
      AND SALARY IN (
            SELECT MIN(SALARY)
            FROM Worker
            WHERE DEPARTMENT='Management'
            AND FIRST_NAME NOT LIKE 'P%'
      )
      AND FIRST_NAME NOT LIKE 'P%';

SELECT MIN(SALARY)
FROM Worker
WHERE DEPARTMENT='Finance';

SELECT MAX(SALARY)
FROM Worker
WHERE DEPARTMENT='Management';

SELECT SUM(SALARY)
FROM Worker
WHERE DEPARTMENT='IT';

SELECT COUNT(WORKER_ID)
FROM Worker
WHERE DEPARTMENT='Support';

SELECT AVG(SALARY)
FROM Worker
WHERE DEPARTMENT='Finance';

SELECT SALARY FROM Worker
UNION
SELECT SALARY FROM Worker;

SELECT SALARY FROM Worker
UNION ALL
SELECT SALARY FROM Worker;

SELECT WORKER_ID AS EMPLOYEE_NUMBER
FROM Worker;

SELECT DISTINCT(DEPARTMENT)
FROM Worker;

SELECT WORKER_ID, FIRST_NAME, DEPARTMENT,
CASE
    WHEN SALARY>400000
            THEN 'Executive Level'
    WHEN SALARY BETWEEN 150000 AND 400000
            THEN 'Mid Level'
    ELSE 'Entry Level'
END
AS SALARY_STATUS
FROM Worker;

SELECT * FROM Worker;

CREATE VIEW LowestPaidEmployees AS
SELECT FIRST_NAME, LAST_NAME, SALARY, DEPARTMENT
FROM Worker
WHERE SALARY IN (
      SELECT MIN(SALARY)
      FROM Worker
      GROUP BY DEPARTMENT
);

CREATE VIEW HighestPaidEmployees AS
SELECT FIRST_NAME, LAST_NAME, SALARY, DEPARTMENT
FROM Worker
WHERE SALARY IN (
      SELECT MAX(SALARY)
      FROM Worker
      GROUP BY DEPARTMENT
);

SELECT *
FROM LowestPaidEmployees;

SELECT *
FROM HighestPaidEmployees;
