DROP TABLE Worker;

CREATE TABLE Worker (
    WORKER_ID INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    FIRST_NAME CHAR(25),
    LAST_NAME CHAR(25),
    SALARY INT(15),
    JOINING_DATE DATETIME,
    DEPARTMENT CHAR(25)
);

SELECT WORKER_ID, FIRST_NAME, LAST_NAME FROM Worker
WHERE DEPARTMENT='Finance' AND SALARY>100000;

SELECT WORKER_ID, FIRST_NAME, LAST_NAME FROM Worker
WHERE DEPARTMENT='IT' OR SALARY>300000;

SELECT * FROM Worker
WHERE SALARY BETWEEN 100000 AND 350000;

SELECT * FROM Worker
WHERE SALARY NOT BETWEEN 100000 AND 350000;

SELECT * FROM Worker
WHERE DEPARTMENT IN ('Support', 'Finance');

SELECT * FROM Worker
WHERE DEPARTMENT NOT IN ('IT', 'Management');

SELECT * FROM Worker
WHERE SALARY BETWEEN 100000 AND 300000
AND DEPARTMENT IN ('IT', 'Finance');

SELECT * FROM Worker
WHERE FIRST_NAME LIKE '_e%';

SELECT MIN(SALARY) FROM Worker
WHERE DEPARTMENT='Finance';

SELECT WORKER_ID, FIRST_NAME,
CASE
    WHEN SALARY>=400000 THEN 'HIGH CLASS'
    WHEN SALARY>=150000 AND SALARY<=399999 THEN 'UPPER MIDDLE'
    ELSE 'NORMAL'
END AS STATUS
FROM Worker;

CREATE VIEW minSalaryFinance AS
SELECT MIN(SALARY) FROM Worker
WHERE DEPARTMENT='Finance';

CREATE VIEW maxSalaryManagement AS
SELECT MAX(SALARY) FROM Worker
WHERE DEPARTMENT='Management';
