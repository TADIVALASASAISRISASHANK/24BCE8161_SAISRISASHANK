 Hid INT PRIMARY KEY,

SELECT * FROM departments NATURAL JOIN HOD;

SELECT * 
FROM departments d
INNER JOIN HOD h 
		ON d.did=h.Hid;
        
SELECT *
FROM departments d
LEFT OUTER JOIN HOD h
		ON 	d.did=h.Hid;
        
SELECT * 
FROM departments d
RIGHT OUTER JOIN HOD h
		ON d.did=h.Hid;
